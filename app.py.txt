import streamlit as st
from openai import OpenAI

st.set_page_config(page_title="My Unlimited ChatGPT", page_icon="🤖")
st.header("Welcome to My Unlimited AI App")

# A secure password-style box where your friends can paste an OpenAI API key
api_key = st.sidebar.text_input("Enter your OpenAI API Key:", type="password")

if api_key:
    client = OpenAI(api_key=api_key)

    if "messages" not in st.session_state:
        st.session_state["messages"] = [{"role": "assistant", "content": "How can I help you today?"}]

    # Display the conversation history on the screen
    for msg in st.session_state["messages"]:
        st.chat_message(msg["role"]).write(msg["content"])

    # Wait for your friend to type a message
    if user_input := st.chat_input("Type your message here..."):
        st.session_state["messages"].append({"role": "user", "content": user_input})
        st.chat_message("user").write(user_input)
        
        # Send it to OpenAI's cheap, uncapped cloud server
        response = client.chat.completions.create(
            model="gpt-4o-mini", 
            messages=st.session_state["messages"]
        )
        
        reply = response.choices[0].message.content
        st.session_state["messages"].append({"role": "assistant", "content": reply})
        st.chat_message("assistant").write(reply)
else:
    st.info("Please enter an OpenAI API Key in the left sidebar to unlock the AI and start chatting!")
